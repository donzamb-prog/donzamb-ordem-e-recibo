function carregarCliente(index) { // 1. Limpa TODOS os campos do formulário primeiro para evitar sobreposição de dados de outros clientes document.getElementById('cli-referencia').value = ''; document.getElementById('cli-nome').value = ''; document.getElementById('cli-email').value = ''; document.getElementById('cli-telefone').value = ''; document.getElementById('cli-cpf').value = ''; document.getElementById('cli-rg').value = ''; document.getElementById('cli-endereco').value = ''; document.getElementById('info-validade').value = ''; document.getElementById('info-marca').value = ''; document.getElementById('info-modelo').value = ''; document.getElementById('info-aparelho').value = ''; document.getElementById('info-equipamento').value = ''; document.getElementById('info-outros').value = ''; document.getElementById('info-acessorios').value = ''; document.getElementById('info-defeito').value = ''; document.getElementById('info-obs').value = ''; document.getElementById('val-visita').value = '0,00'; document.getElementById('desc-visita').value = '0,00'; document.getElementById('desc-pecas').value = '0,00'; document.getElementById('desc-servicos').value = '0,00'; document.querySelectorAll('.input-taxa').forEach(input => input.value = '0,00'); document.getElementById('pag-vista').value = ''; document.getElementById('pag-sinal').value = ''; document.getElementById('pag-parcelas').value = ''; document.querySelectorAll('.meio-pag').forEach(chk => chk.checked = false); document.getElementById('add-garantia').value = '90 dias conforme termo'; document.getElementById('add-anotacoes').value = ''; document.getElementById('add-relatorio').value = ''; document.getElementById('add-obs').value = ''; document.getElementById('container-pecas').innerHTML = ''; document.getElementById('container-servicos').innerHTML = ''; countPeca = 0; countServico = 0; if (index === '') { adicionarPeca(); adicionarServico(); calcularTotais(); return; } let c = clientesGlobais[index]; if (!c) return; // 2. Preenche os Dados Cadastrais do Cliente Selecionado document.getElementById('cli-referencia').value = c.ref || ''; document.getElementById('cli-nome').value = c.nome || ''; document.getElementById('cli-email').value = c.email || ''; document.getElementById('cli-telefone').value = c.tel || ''; document.getElementById('cli-cpf').value = c.cpf || ''; document.getElementById('cli-rg').value = c.rg || ''; document.getElementById('cli-endereco').value = c.end || ''; // 3. Busca o ÚLTIMO PEDIDO feito para este cliente let ultimoPedido = null; for (let i = pedidosGlobais.length - 1; i >= 0; i--) { if (pedidosGlobais[i].cliente && pedidosGlobais[i].cliente.trim().toLowerCase() === c.nome.trim().toLowerCase()) { ultimoPedido = pedidosGlobais[i]; break; } } // 4. Se encontrar um pedido anterior, carrega todos os dados if (ultimoPedido) { document.getElementById('info-marca').value = ultimoPedido.marca || ''; document.getElementById('info-modelo').value = ultimoPedido.modelo || ''; document.getElementById('info-aparelho').value = ultimoPedido.aparelho || ''; document.getElementById('info-acessorios').value = ultimoPedido.acessorios || ''; document.getElementById('info-defeito').value = ultimoPedido.defeito || ''; // Recarrega Peças if (ultimoPedido.pecas) { let listaPecas = Array.isArray(ultimoPedido.pecas) ? ultimoPedido.pecas : ultimoPedido.pecas.split(' | '); listaPecas.forEach(itemStr => { if (!itemStr.trim()) return; adicionarPeca(); let row = document.getElementById(`peca-row-${countPeca}`); if (row) { let qtdMatch = itemStr.match(/^(\d+)x\s+/); let qtd = qtdMatch ? qtdMatch[1] : '1'; let valMatch = itemStr.match(/\(R\$\s*([\d.,]+)\)/); let val = valMatch ? valMatch[1] : '0,00'; let desc = itemStr.replace(/^(\d+)x\s+/, '').replace(/\s+\(R\$\s*[\d.,]+\)/, '').trim(); row.querySelector('.peca-qtd').value = qtd; row.querySelector('.peca-desc').value = desc; row.querySelector('.peca-val').value = val; } }); } // Recarrega Serviços if (ultimoPedido.servicos) { let listaServicos = Array.isArray(ultimoPedido.servicos) ? ultimoPedido.servicos : ultimoPedido.servicos.split(' | '); listaServicos.forEach(itemStr => { if (!itemStr.trim()) return; adicionarServico(); let row = document.getElementById(`servico-row-${countServico}`); if (row) { let valMatch = itemStr.match(/\(R\$\s*([\d.,]+)\)/); let val = valMatch ? valMatch[1] : '0,00'; let desc = itemStr.replace(/\s+\(R\$\s*[\d.,]+\)/, '').trim(); row.querySelector('.serv-desc').value = desc; row.querySelector('.serv-val').value = val; } }); } // Recarrega Informações Financeiras, Relatórios e Garantia if (ultimoPedido.garantia) document.getElementById('add-garantia').value = ultimoPedido.garantia; if (ultimoPedido.sinal) document.getElementById('pag-sinal').value = formatarMoeda(ultimoPedido.sinal); if (ultimoPedido.parcelas) document.getElementById('pag-parcelas').value = ultimoPedido.parcelas; if (ultimoPedido.relatorio) document.getElementById('add-relatorio').value = ultimoPedido.relatorio; if (ultimoPedido.anotacoes) document.getElementById('add-anotacoes').value = ultimoPedido.anotacoes; if (ultimoPedido.obs) document.getElementById('add-obs').value = ultimoPedido.obs; // Garante linhas padrão caso não haja peças ou serviços if (countPeca === 0) adicionarPeca(); if (countServico === 0) adicionarServico(); calcularTotais(); } else { adicionarPeca(); adicionarServico(); calcularTotais(); } }
